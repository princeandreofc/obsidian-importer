/* ═══════════ ZIP: escrita (store) ═══════════ */
const CRC_T=(()=>{const t=new Uint32Array(256);for(let n=0;n<256;n++){let c=n;
  for(let k=0;k<8;k++)c=(c&1)?(0xEDB88320^(c>>>1)):(c>>>1);t[n]=c>>>0;}return t;})();
function crc32(b){let c=0xFFFFFFFF;for(let i=0;i<b.length;i++)c=CRC_T[(c^b[i])&0xFF]^(c>>>8);return (c^0xFFFFFFFF)>>>0;}

function zipWrite(arqs){
  const loc=[],cen=[];let off=0;
  for(const a of arqs){
    const nome=utf8(a.nome),d=a.dados,crc=crc32(d);
    const lh=new DataView(new ArrayBuffer(30));
    lh.setUint32(0,0x04034b50,true);lh.setUint16(4,20,true);lh.setUint16(6,0x0800,true);
    lh.setUint16(8,0,true);lh.setUint32(14,crc,true);
    lh.setUint32(18,d.length,true);lh.setUint32(22,d.length,true);lh.setUint16(26,nome.length,true);
    loc.push(new Uint8Array(lh.buffer),nome,d);
    const ch=new DataView(new ArrayBuffer(46));
    ch.setUint32(0,0x02014b50,true);ch.setUint16(4,20,true);ch.setUint16(6,20,true);
    ch.setUint16(8,0x0800,true);ch.setUint16(10,0,true);ch.setUint32(16,crc,true);
    ch.setUint32(20,d.length,true);ch.setUint32(24,d.length,true);ch.setUint16(28,nome.length,true);
    ch.setUint32(42,off,true);
    cen.push(new Uint8Array(ch.buffer),nome);
    off+=30+nome.length+d.length;
  }
  const tc=cen.reduce((s,b)=>s+b.length,0);
  const fim=new DataView(new ArrayBuffer(22));
  fim.setUint32(0,0x06054b50,true);fim.setUint16(8,arqs.length,true);fim.setUint16(10,arqs.length,true);
  fim.setUint32(12,tc,true);fim.setUint32(16,off,true);
  const partes=[...loc,...cen,new Uint8Array(fim.buffer)];
  const tot=partes.reduce((s,b)=>s+b.length,0),out=new Uint8Array(tot);
  let p=0;for(const b of partes){out.set(b,p);p+=b.length;}
  return out;
}

/* ═══════════ ZIP: leitura (store + deflate) ═══════════ */
async function zipRead(buf){
  const dv=new DataView(buf),u8=new Uint8Array(buf);
  let eocd=-1;
  for(let i=u8.length-22;i>=0;i--){ if(dv.getUint32(i,true)===0x06054b50){eocd=i;break;} }
  if(eocd<0) throw new Error("Não parece um .xlsx válido.");
  const n=dv.getUint16(eocd+10,true);
  let p=dv.getUint32(eocd+16,true);
  const saida=[];
  for(let i=0;i<n;i++){
    if(dv.getUint32(p,true)!==0x02014b50) throw new Error("Diretório do arquivo corrompido.");
    const metodo=dv.getUint16(p+10,true);
    const compSize=dv.getUint32(p+20,true);
    const nomeLen=dv.getUint16(p+28,true);
    const extraLen=dv.getUint16(p+30,true);
    const cmtLen=dv.getUint16(p+32,true);
    const localOff=dv.getUint32(p+42,true);
    const nome=new TextDecoder().decode(u8.subarray(p+46,p+46+nomeLen));
    // cabeçalho local tem tamanhos próprios de nome/extra
    const lNomeLen=dv.getUint16(localOff+26,true);
    const lExtraLen=dv.getUint16(localOff+28,true);
    const ini=localOff+30+lNomeLen+lExtraLen;
    let dados=u8.subarray(ini,ini+compSize);
    if(metodo===8){
      const ds=new DecompressionStream("deflate-raw");
      const blob=new Blob([dados]).stream().pipeThrough(ds);
      dados=new Uint8Array(await new Response(blob).arrayBuffer());
    } else if(metodo!==0){
      throw new Error("Compressão não suportada no arquivo modelo.");
    }
    saida.push({nome,dados});
    p+=46+nomeLen+extraLen+cmtLen;
  }
  return saida;
}

/* ═══════════ XLSX ═══════════ */
/** Letra da coluna a partir do índice 0. Passa de Z: 26 -> AA. */
function colLetra(i){
  let s="",n=i+1;
  while(n>0){ const r=(n-1)%26; s=String.fromCharCode(65+r)+s; n=Math.floor((n-1)/26); }
  return s;
}

function linhasXml(linhas,inicio){
  return linhas.map((linha,idx)=>{
    const r=inicio+idx;
    const cel=linha.map((v,ci)=>{
      const col=colLetra(ci);
      if(v===null||v===undefined||v==="") return "";
      return typeof v==="number"
        ? `<c r="${col}${r}"><v>${v}</v></c>`
        : `<c r="${col}${r}" t="inlineStr"><is><t xml:space="preserve">${esc(v)}</t></is></c>`;
    }).join("");
    return `<row r="${r}">${cel}</row>`;
  }).join("");
}

/* --- modelo do Mercos: preserva tudo, só acrescenta linhas na 1ª aba --- */
let MODELO=null;   // {partes, alvo, cabecalho, primeiraLinhaVazia}

function alvoDaPrimeiraAba(partes){
  // a 1ª aba é a que o workbook lista primeiro; resolve pelo rels
  const dec=new TextDecoder();
  const wb=partes.find(p=>p.nome==="xl/workbook.xml");
  const rel=partes.find(p=>p.nome==="xl/_rels/workbook.xml.rels");
  if(!wb||!rel) return "xl/worksheets/sheet1.xml";
  const rid=(dec.decode(wb.dados).match(/<sheet[^>]*r:id="([^"]+)"/)||[])[1];
  if(!rid) return "xl/worksheets/sheet1.xml";
  const alvo=(dec.decode(rel.dados)
    .match(new RegExp(`Id="${rid}"[^>]*Target="([^"]+)"`))||[])[1];
  if(!alvo) return "xl/worksheets/sheet1.xml";
  return "xl/" + alvo.replace(/^\/?xl\//,"").replace(/^\//,"");
}

function lerCabecalho(sheetXml,partes){
  const dec=new TextDecoder();
  // strings compartilhadas, se houver
  const ss=partes.find(p=>p.nome==="xl/sharedStrings.xml");
  let tabela=[];
  if(ss){
    tabela=[...dec.decode(ss.dados).matchAll(/<si>([\s\S]*?)<\/si>/g)]
      .map(m=>[...m[1].matchAll(/<t[^>]*>([\s\S]*?)<\/t>/g)].map(x=>x[1]).join(""));
  }
  const linha1=(sheetXml.match(/<row[^>]*\br="1"[^>]*>([\s\S]*?)<\/row>/)||[])[1]||"";
  // O atributo t vem em qualquer posição do <c>, e a célula pode ser autofechada.
  // Por isso os atributos são lidos separados do corpo.
  const cols=[...linha1.matchAll(/<c\b([^>]*?)(?:\/>|>([\s\S]*?)<\/c>)/g)].map(m=>{
    const attrs=m[1]||"", corpo=m[2]||"";
    const tipo=(attrs.match(/\bt="([^"]+)"/)||[])[1];
    if(tipo==="s"){                                   // string compartilhada
      const i=+((corpo.match(/<v>(\d+)<\/v>/)||[])[1]);
      return tabela[i]??"";
    }
    const t=corpo.match(/<t[^>]*>([\s\S]*?)<\/t>/);   // inlineStr, e também str
    if(t) return t[1];
    const v=corpo.match(/<v>([\s\S]*?)<\/v>/);
    return v?v[1]:"";
  });
  return cols.map(desescapar);
}

const desescapar=s=>String(s).replace(/&lt;/g,"<").replace(/&gt;/g,">")
  .replace(/&quot;/g,'"').replace(/&apos;/g,"'").replace(/&amp;/g,"&");

  }

/** Casa os campos do catálogo com as colunas do modelo, pelo nome do cabeçalho. */
function mapearColuna(nomeColuna){
  const n=nomeColuna.toLowerCase();
  if(/c[óo]digo/.test(n)) return p=>p.cod;
  if(/descri[çc][ãa]o|nome|produto/.test(n)) return p=>p.nome;
  if(/pre[çc]o|valor/.test(n)) return p=>p.preco;
  return ()=>"";                       // coluna não reconhecida fica vazia
}

async function baixar(nome,bytes){
  const d=await (window.claude?.use?.("downloads") ?? Promise.resolve(null));
  if(!d){aviso("Download indisponível nesta visualização.");return false;}
  try{ await d.save({filename:nome,data:bytes}); aviso("Planilha gerada."); return true; }
  catch(e){ aviso("Download não concluído."); return false; }
}
